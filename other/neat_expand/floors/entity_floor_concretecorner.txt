/* CookieServerPatch override of Neat Building floor (same file path, loads last).
   The original rows used the map's own ground tiles (blends_natural / blends_street),
   so every natural ground tile in newly loaded areas became an entity (server lag).
   Same entity names (saves keep loading), rows point at CookieServerPatch's own
   unused tiles csp_nofloor_*, and the recipe is hidden from the build menu. */
module Base
{
	xuiSkin default
	{
		entity ES_Floor_ConcreteCorner
		{
			LuaWindowClass	= ISEntityWindow,
			DisplayName	= NBEntityName_ConcreteCornerFloor,
			Icon		= Build_blends_street_01_99,
		}
	}

    entity Floor_ConcreteCorner
    {
        component UiConfig
                {
                    xuiSkin         = default,
                    entityStyle     = ES_Floor_ConcreteCorner,
                    uiEnabled       = false,
                }

        component SpriteConfig
        {
		    isThumpable     = false,
		    OnIsValid       = BuildRecipeCode.floor.OnIsValid,
		    OnCreate        = NB_BuildRecipeCode.PartFloors.OnCreate,
			
            face W
            {
                layer
                {
                    row = csp_nofloor_40,
                }
            }
            face N
            {
                layer
                {
                    row = csp_nofloor_41,
                }
            }
            face E
            {
                layer
                {
                    row = csp_nofloor_42,
                }
            }
            face S
            {
                layer
                {
                    row = csp_nofloor_43,
                }
            }
        }
		
	component CraftRecipe
        {
            OnAddToMenu   = ignoreFromBuildMenu,
            timedAction   = BuildLowNoTool,
            time          = 100,
            category      = Floors,
            ToolTip       = Tooltip_craft_sandFloorDesc,
            inputs
            {
                item 1 tags[Concrete] flags[DontRecordInput],
            }
        }
    }
}
